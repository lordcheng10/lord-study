02

state.py + builder.py

# state.py
from typing import Annotated, TypedDict
from operator import add

class AgentState(TypedDict, total=False):
    input: str
    output: str
    retry_count: Annotated[int, add]  # Critic retry
    triage_result: dict   # 各 SP 各占独立字段，避免并行 Reducer 冲突
# builder.py · LangGraph 装配
from langgraph.graph import END, StateGraph

def build_graph():
    g = StateGraph(AgentState)
    g.add_node("scan", scan_node)
    g.add_node("process", process_node)
    g.set_entry_point("scan")
    g.add_edge("scan", "process")
    g.add_edge("process", END)
    return g.compile()

多 SP 并行扇出/扇入 → 参考 sre-rca-agent/builder.py 的 recall → 4 specialists → rag → synthesizer → critic 链路。

from ai_trust_toolkit.specialists import sp_node
from ai_trust_toolkit.llm import get_llm

@sp_node(name="triage", prompt_file="prompts/triage.md")
async def triage(state: AgentState) -> dict:
    llm = get_llm(tier="auto")  # cost-aware · 别 hardcode "sonnet"
    result = await llm.ainvoke(state["input"])
    return {"category": parse_category(result.content)}
