from operator import add

class AgentState(TypedDict, total=False):
    input: str
    output: str
    retry_count: Annotated[int, add]

③ my_agent/nodes/__init__.py

from .run import run_node
__all__ = ["run_node"]

④ my_agent/builder.py — 与 nodes/ 平级，负责串图

from langgraph.graph import END, StateGraph
from .nodes import run_node
from .state import AgentState

def build_graph():
    g = StateGraph(AgentState)
    g.add_node("run", run_node)
    g.set_entry_point("run")
    g.add_edge("run", END)
    return g.compile()
