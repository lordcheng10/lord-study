常见误解：builder.py 在 <repo>/my_agent/builder.py，与 nodes/ 平级。nodes/ 里只放节点函数（如 run.py），没有 builder.py。

# <repo>/my_agent/builder.py · 串图示例（文件在 my_agent/ 根下，不是 nodes/ 下）
from langgraph.graph import END, StateGraph
from .nodes import run_node          # 从 nodes/  import 进来
from .state import AgentState

def build_graph():
    g = StateGraph(AgentState)
    g.add_node("run", run_node)
    g.set_entry_point("run")
    g.add_edge("run", END)
    return g.compile()
